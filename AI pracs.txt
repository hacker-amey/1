% --- BFS search loop ---
bfs_search([[Goal|RestPath] | _], Goal, [Goal|RestPath]). % Goal found
bfs_search([[Node|RestPath] | OtherPaths], Goal, Path) :-
 findall([Next,Node|RestPath],
 (edge(Node, Next), \+ member(Next, [Node|RestPath])),
 NewPaths),
 append(OtherPaths, NewPaths, UpdatedQueue),
 bfs_search(UpdatedQueue, Goal, Path).

%% Public entry
solve_best_first(Start, Goal, Path) :-
 h_misplaced(Start, Goal, H0),
 best_first([H0-(Start-[Start])], Goal, [], Path). % open: list of H-(State-PathRev)
%% Main loop
best_first([], _, _, _) :- % no solution
 fail.
best_first([_H-(State-PathRev)|_], Goal, _, Path) :-
 State == Goal, % goal test
 reverse(PathRev, Path), !. % return path (start..goal)
best_first([_H-(State-PathRev)|RestOpen], Goal, Closed, Path) :-
 findall(Hn-(Next-NewPathRev),
 ( move(State, Next),
 \+ member(Next, PathRev), % avoid cycles by path check
 \+ member(Next, Closed), % optional: avoid states already closed
 h_misplaced(Next, Goal, Hn),
 NewPathRev = [Next|PathRev]
 ),
 Children),
 append(RestOpen, Children, Open2),
 keysort(Open2, SortedOpen), % sort by heuristic (smallest H first)
 best_first(SortedOpen, Goal, [State|Closed], Path).

% --- Alpha-Beta Pruning (minimal version) ---

alpha_beta(Pos, _, _, _, Val) :-
    terminal(Pos), value(Pos, Val), !.

alpha_beta(Pos, A, B, max, Val) :-
    findall(M, move(Pos, M), Ms),
    best(Ms, Pos, A, B, max, -inf, Val).

alpha_beta(Pos, A, B, min, Val) :-
    findall(M, move(Pos, M), Ms),
    best(Ms, Pos, A, B, min, inf, Val).

best([], _, _, _, _, V, V).
best([M|Ms], Pos, A, B, max, Cur, Val) :-
    apply(Pos, M, P1),
    alpha_beta(P1, A, B, min, V),
    A1 is max(A, V),
    ( A1 >= B -> Val = A1
    ; max(Cur, V, C1),
      best(Ms, Pos, A1, B, max, C1, Val)
    ).

best([M|Ms], Pos, A, B, min, Cur, Val) :-
    apply(Pos, M, P1),
    alpha_beta(P1, A, B, max, V),
    B1 is min(B, V),
    ( A >= B1 -> Val = B1
    ; min(Cur, V, C1),
      best(Ms, Pos, A, B1, min, C1, Val)
    ).

% Main predicate – just the solving logic
solve_and_show :-
    queens(Board),
    write('Solution: '), write(Board), nl.

% queens(Qs) – Qs is a list of 8 integers; each represents the row
% position of the queen in that column.
queens(Qs) :-
    permutation([1,2,3,4,5,6,7,8], Qs),
    safe(Qs).

% safe(List) – no two queens attack each other
safe([]).
safe([Q|Qs]) :-
    no_attack(Q, Qs, 1),
    safe(Qs).

% no_attack(Q, Rest, D) – Q doesn’t attack any queen in Rest.
% D is the column distance between Q and the queens in Rest.
no_attack(_, [], _).
no_attack(Q, [Q1|Qs], D) :-
    Q =\= Q1,            % different rows
    abs(Q - Q1) =\= D,   % not on same diagonal
    D1 is D + 1,
    no_attack(Q, Qs, D1).
